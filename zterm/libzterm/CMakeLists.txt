cmake_minimum_required(VERSION 3.25)
project(libzterm VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(libzterm src/terminal.cpp)
add_library(libzterm::terminal ALIAS libzterm)
set_target_properties(libzterm PROPERTIES OUTPUT_NAME zterm EXPORT_NAME terminal)
target_compile_features(libzterm PUBLIC cxx_std_23)
target_compile_options(libzterm PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(libzterm PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
install(TARGETS libzterm EXPORT libztermTargets LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}" ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(EXPORT libztermTargets NAMESPACE libzterm:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/libzterm")
configure_package_config_file(libztermConfig.cmake.in libztermConfig.cmake INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/libzterm")
configure_file(libzterm.pc.in libzterm.pc @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libztermConfig.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/libzterm")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libzterm.pc" DESTINATION "${CMAKE_INSTALL_LIBDIR}/pkgconfig")
